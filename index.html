/** * ============================================================ * Multi-Panel Monitoring Backend — Google Apps Script * ============================================================ * Mendukung multiple panel (ION9000, panel lain, dll). * Setiap panel punya spreadsheet sendiri, di-rotate per bulan. * * Data dari ESP32 dikirim dengan field "panelId" (mis: "panel1") * Kalau tidak ada panelId, default "panel1". * ============================================================ */ // ============ KONFIGURASI ============ const DRIVE_FOLDER_ID = 'GANTI_DENGAN_FOLDER_ID_ANDA'; const RETENTION_MONTHS = 3; const CACHE_SECONDS = 8; const TOKEN_EXPIRY_DAYS = 7; const DEFAULT_PANEL_ID = 'panel1'; // ⚠️ GANTI USERNAME & PASSWORD INI! const USERS = { 'admin': 'ion9000admin', 'operator': 'operator123', 'viewer': 'viewer123' }; // ============ SCRIPT PROPERTIES ============ const PROP_TOKENS = 'auth_tokens'; const PROP_PANEL_INDEX = 'panel_index'; // JSON: {panelId: {name, activeMonth, activeSSId, months: [{month,id,name}]}} // ============ ENTRY POINT: POST ============ function doPost(e) { try { if (!e || !e.postData || !e.postData.contents) { return jsonResponse({ status: 'error', message: 'No POST data' }); } const body = JSON.parse(e.postData.contents); if (body.action === 'login') { return handleLogin(body); } // Data sensor — cek panelId const panelId = body.panelId || DEFAULT_PANEL_ID; // Buang panelId dari data sebelum disimpan (biar tidak jadi kolom) const data = Object.assign({}, body); delete data.panelId; delete data.action; const result = writeToSheet(panelId, data); return jsonResponse({ status: 'success', message: 'Data tersimpan', panelId: panelId, row: result.row, sheet: result.sheetName, timestamp: result.timestamp }); } catch (err) { console.error('doPost error:', err); return jsonResponse({ status: 'error', message: err.message }); } } // ============ ENTRY POINT: GET ============ function doGet(e) { try { const action = (e.parameter.action || 'realtime').toLowerCase(); if (action === 'health') { return jsonResponse({ status: 'ok', time: new Date().toISOString() }); } if (action === 'login') { return handleLogin(e.parameter); } const token = e.parameter.token; const user = validateToken(token); if (!user) { return jsonResponse({ status: 'error', message: 'Unauthorized' }, 401); } const panelId = e.parameter.panelId || DEFAULT_PANEL_ID; switch (action) { case 'realtime': return jsonResponse(getRealtime(panelId)); case 'latest': return jsonResponse(getLatest(panelId, parseInt(e.parameter.limit) || 100)); case 'history': return jsonResponse(getHistory(panelId, e.parameter.date)); case 'months': return jsonResponse(getAvailableMonths(panelId)); case 'daily_summary': return jsonResponse(getDailySummary(panelId, e.parameter.date)); case 'chart': return jsonResponse(getChartData(panelId, e.parameter.period, e.parameter.param, e.parameter.date)); case 'cb_timeline': return jsonResponse(getCbTimeline(panelId, e.parameter.period, e.parameter.date)); case 'panels': return jsonResponse(listPanels()); case 'check': return jsonResponse({ status: 'success', user: user }); default: return jsonResponse({ status: 'error', message: 'Unknown action' }); } } catch (err) { console.error('doGet error:', err); return jsonResponse({ status: 'error', message: err.message }, 500); } } // ============ LOGIN ============ function handleLogin(params) { const username = params.username; const password = params.password; if (!username || !password) { return jsonResponse({ status: 'error', message: 'Username & password wajib diisi' }); } if (!USERS[username] || USERS[username] !== password) { Utilities.sleep(500); return jsonResponse({ status: 'error', message: 'Username atau password salah' }); } const token = generateToken(); const expires = Date.now() + (TOKEN_EXPIRY_DAYS * 24 * 60 * 60 * 1000); const props = PropertiesService.getScriptProperties(); let tokens = {}; try { tokens = JSON.parse(props.getProperty(PROP_TOKENS) || '{}'); } catch (e) { tokens = {}; } const now = Date.now(); Object.keys(tokens).forEach(t => { if (tokens[t].expires < now) delete tokens[t]; }); tokens[token] = { user: username, expires: expires }; props.setProperty(PROP_TOKENS, JSON.stringify(tokens)); return jsonResponse({ status: 'success', token: token, user: username, expires: expires }); } function generateToken() { const bytes = []; for (let i = 0; i < 32; i++) bytes.push(Math.floor(Math.random() * 256)); return Utilities.base64EncodeWebSafe(bytes); } function validateToken(token) { if (!token) return null; const props = PropertiesService.getScriptProperties(); let tokens = {}; try { tokens = JSON.parse(props.getProperty(PROP_TOKENS) || '{}'); } catch (e) { return null; } const entry = tokens[token]; if (!entry) return null; if (entry.expires < Date.now()) { delete tokens[token]; props.setProperty(PROP_TOKENS, JSON.stringify(tokens)); return null; } return { username: entry.user }; } // ============ PANEL INDEX MANAGEMENT ============ function getPanelIndex() { const props = PropertiesService.getScriptProperties(); try { return JSON.parse(props.getProperty(PROP_PANEL_INDEX) || '{}'); } catch (e) { return {}; } } function savePanelIndex(index) { const props = PropertiesService.getScriptProperties(); props.setProperty(PROP_PANEL_INDEX, JSON.stringify(index)); } function ensurePanel(panelId) { const index = getPanelIndex(); if (!index[panelId]) { index[panelId] = { name: panelId.charAt(0).toUpperCase() + panelId.slice(1), activeMonth: null, activeSSId: null, months: [] }; savePanelIndex(index); } return index[panelId]; } function listPanels() { const index = getPanelIndex(); const panels = Object.keys(index).map(id => ({ id: id, name: index[id].name, activeMonth: index[id].activeMonth, monthsCount: (index[id].months || []).length })); return { status: 'success', data: panels }; } // ============ TULIS DATA ============ function writeToSheet(panelId, data) { const now = new Date(); const monthKey = Utilities.formatDate(now, 'Asia/Jakarta', 'yyyy-MM'); const timestamp = Utilities.formatDate(now, 'Asia/Jakarta', 'yyyy-MM-dd HH:mm:ss'); const ss = getOrCreateMonthlySpreadsheet(panelId, monthKey); const sheet = ss.getSheetByName('Data') || ss.insertSheet('Data'); const headers = getOrCreateHeaders(sheet, data); const row = headers.map(h => { if (h === 'timestamp') return timestamp; return data[h] !== undefined && data[h] !== null ? data[h] : ''; }); sheet.appendRow(row); cleanupOldSpreadsheets(panelId); return { row: sheet.getLastRow(), sheetName: ss.getName(), timestamp: timestamp }; } function getOrCreateHeaders(sheet, data) { if (sheet.getLastRow() === 0) { const headers = ['timestamp'].concat(Object.keys(data)); sheet.appendRow(headers); sheet.getRange(1, 1, 1, headers.length) .setFontWeight('bold') .setBackground('#4CAF50') .setFontColor('#FFFFFF'); sheet.setFrozenRows(1); return headers; } return sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; } // ============ SPREADSHEET BULANAN (per panel) ============ function getOrCreateMonthlySpreadsheet(panelId, monthKey) { const index = getPanelIndex(); const panel = index[panelId]; if (panel && panel.activeMonth === monthKey && panel.activeSSId) { try { return SpreadsheetApp.openById(panel.activeSSId); } catch (err) { console.warn('Active spreadsheet not found: ' + err.message); } } const ssName = 'Monitor_' + panelId + '_' + monthKey; const ss = SpreadsheetApp.create(ssName); if (DRIVE_FOLDER_ID && DRIVE_FOLDER_ID !== 'GANTI_DENGAN_FOLDER_ID_ANDA') { try { const file = DriveApp.getFileById(ss.getId()); const folder = DriveApp.getFolderById(DRIVE_FOLDER_ID); folder.addFile(file); DriveApp.getRootFolder().removeFile(file); } catch (err) { console.warn('Cannot move to folder: ' + err.message); } } ss.getSheets()[0].setName('Data'); // Update index const idx = getPanelIndex(); if (!idx[panelId]) { idx[panelId] = { name: panelId, activeMonth: null, activeSSId: null, months: [] }; } idx[panelId].activeMonth = monthKey; idx[panelId].activeSSId = ss.getId(); idx[panelId].months = (idx[panelId].months || []).filter(m => m.month !== monthKey); idx[panelId].months.push({ month: monthKey, id: ss.getId(), name: ssName }); idx[panelId].months.sort((a, b) => b.month.localeCompare(a.month)); savePanelIndex(idx); return ss; } // ============ RETENSI (per panel) ============ function cleanupOldSpreadsheets(panelId) { const idx = getPanelIndex(); const panel = idx[panelId]; if (!panel || !panel.months || panel.months.length <= RETENTION_MONTHS) return; panel.months.sort((a, b) => b.month.localeCompare(a.month)); const toKeep = panel.months.slice(0, RETENTION_MONTHS); const toDelete = panel.months.slice(RETENTION_MONTHS); toDelete.forEach(item => { try { DriveApp.getFileById(item.id).setTrashed(true); console.log('Trashed: ' + item.name); } catch (err) { console.warn('Cannot trash ' + item.name + ': ' + err.message); } }); panel.months = toKeep; savePanelIndex(idx); } // ============ HELPER: BUKA SHEET PANEL ============ function getPanelSheet(panelId, monthKey) { const idx = getPanelIndex(); const panel = idx[panelId]; if (!panel) return null; let entry; if (monthKey) { entry = (panel.months || []).find(m => m.month === monthKey); } else { entry = (panel.months || []).find(m => m.month === panel.activeMonth); } if (!entry) return null; try { const ss = SpreadsheetApp.openById(entry.id); return { ss: ss, sheet: ss.getSheetByName('Data'), entry: entry }; } catch (err) { return null; } } // ============ API: REALTIME ============ function getRealtime(panelId) { const cache = CacheService.getScriptCache(); const cacheKey = 'realtime_' + panelId; const cached = cache.get(cacheKey); if (cached) return JSON.parse(cached); const info = getPanelSheet(panelId); if (!info || !info.sheet || info.sheet.getLastRow() < 2) { return { status: 'success', data: null, panelId: panelId }; } const sheet = info.sheet; const lastRow = sheet.getLastRow(); const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; const values = sheet.getRange(lastRow, 1, 1, headers.length).getValues()[0]; const obj = {}; headers.forEach((h, i) => obj[h] = values[i]); const result = { status: 'success', data: obj, panelId: panelId }; cache.put(cacheKey, JSON.stringify(result), CACHE_SECONDS); return result; } // ============ API: LATEST ============ function getLatest(panelId, limit) { const info = getPanelSheet(panelId); if (!info || !info.sheet || info.sheet.getLastRow() < 2) { return { status: 'success', data: [], panelId: panelId }; } const sheet = info.sheet; const lastRow = sheet.getLastRow(); const startRow = Math.max(2, lastRow - limit + 1); const numRows = lastRow - startRow + 1; const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; const values = sheet.getRange(startRow, 1, numRows, headers.length).getValues(); const data = values.map(row => { const obj = {}; headers.forEach((h, i) => obj[h] = row[i]); return obj; }); return { status: 'success', data: data, panelId: panelId }; } // ============ API: HISTORY ============ function getHistory(panelId, dateStr) { if (!dateStr) return { status: 'error', message: 'date required' }; const monthKey = dateStr.substring(0, 7); const info = getPanelSheet(panelId, monthKey); if (!info || !info.sheet || info.sheet.getLastRow() < 2) { return { status: 'success', data: [], panelId: panelId }; } const sheet = info.sheet; const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; const allValues = sheet.getRange(2, 1, sheet.getLastRow() - 1, headers.length).getValues(); const filtered = allValues.filter(row => String(row[0]).startsWith(dateStr)); const data = filtered.map(row => { const obj = {}; headers.forEach((h, i) => obj[h] = row[i]); return obj; }); return { status: 'success', data: data, count: data.length, panelId: panelId }; } // ============ API: MONTHS ============ function getAvailableMonths(panelId) { const idx = getPanelIndex(); const panel = idx[panelId]; return { status: 'success', data: (panel && panel.months) || [], panelId: panelId }; } // ============ API: DAILY SUMMARY ============ function getDailySummary(panelId, dateStr) { const hist = getHistory(panelId, dateStr); if (!hist.data || hist.data.length === 0) { return { status: 'success', data: null, panelId: panelId }; } const rows = hist.data; const numericKeys = Object.keys(rows[0]).filter(k => k !== 'timestamp' && k !== 'cb_status'); const summary = {}; numericKeys.forEach(key => { const nums = rows.map(r => parseFloat(r[key])).filter(n => !isNaN(n)); if (nums.length === 0) return; summary[key] = { min: Math.min.apply(null, nums), max: Math.max.apply(null, nums), avg: nums.reduce((a, b) => a + b, 0) / nums.length }; }); return { status: 'success', data: summary, count: rows.length, panelId: panelId }; } // ============ API: CHART DATA ============ function getChartData(panelId, period, param, dateStr) { if (!period || !param) { return { status: 'error', message: 'period & param required' }; } if (!dateStr) { dateStr = Utilities.formatDate(new Date(), 'Asia/Jakarta', 'yyyy-MM-dd'); } const endDate = new Date(dateStr + 'T23:59:59+07:00'); let startDate = new Date(endDate); let bucketMinutes = 5; if (period === '1d') { startDate.setDate(startDate.getDate() - 1); bucketMinutes = 5; } else if (period === '1w') { startDate.setDate(startDate.getDate() - 7); bucketMinutes = 60; } else if (period === '1m') { startDate.setDate(startDate.getDate() - 30); bucketMinutes = 180; } else { return { status: 'error', message: 'period harus 1d, 1w, atau 1m' }; } const startMonth = Utilities.formatDate(startDate, 'Asia/Jakarta', 'yyyy-MM'); const endMonth = Utilities.formatDate(endDate, 'Asia/Jakarta', 'yyyy-MM'); const idx = getPanelIndex(); const panel = idx[panelId]; if (!panel || !panel.months) return { status: 'success', data: [], panelId: panelId }; const months = panel.months .filter(m => m.month >= startMonth && m.month <= endMonth) .map(m => m.month); const allRows = []; months.forEach(m => { const info = getPanelSheet(panelId, m); if (!info || !info.sheet) return; try { const sheet = info.sheet; const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; const tsIdx = headers.indexOf('timestamp'); const paramIdx = headers.indexOf(param); if (tsIdx === -1 || paramIdx === -1) return; const values = sheet.getRange(2, 1, sheet.getLastRow() - 1, headers.length).getValues(); values.forEach(row => { const ts = new Date(String(row[tsIdx]).replace(' ', 'T') + '+07:00'); if (isNaN(ts.getTime())) return; if (ts < startDate || ts > endDate) return; const val = parseFloat(row[paramIdx]); if (isNaN(val)) return; allRows.push({ ts: ts.getTime(), val: val }); }); } catch (err) { console.warn('Skip ' + m + ': ' + err.message); } }); allRows.sort((a, b) => a.ts - b.ts); const bucketMs = bucketMinutes * 60 * 1000; const buckets = {}; allRows.forEach(r => { const key = Math.floor(r.ts / bucketMs) * bucketMs; if (!buckets[key]) buckets[key] = []; buckets[key].push(r.val); }); const data = Object.keys(buckets).sort().map(key => { const vals = buckets[key]; return { t: parseInt(key), avg: vals.reduce((a, b) => a + b, 0) / vals.length, min: Math.min.apply(null, vals), max: Math.max.apply(null, vals), count: vals.length }; }); return { status: 'success', data: data, panelId: panelId, meta: { period: period, param: param, start: startDate.toISOString(), end: endDate.toISOString(), bucketMinutes: bucketMinutes, totalRaw: allRows.length, totalBuckets: data.length } }; } // ============ API: CB TIMELINE ============ function getCbTimeline(panelId, period, dateStr) { if (!period) period = '1d'; if (!dateStr) { dateStr = Utilities.formatDate(new Date(), 'Asia/Jakarta', 'yyyy-MM-dd'); } const endDate = new Date(dateStr + 'T23:59:59+07:00'); let startDate = new Date(endDate); if (period === '1d') { startDate.setDate(startDate.getDate() - 1); } else if (period === '1w') { startDate.setDate(startDate.getDate() - 7); } else if (period === '1m') { startDate.setDate(startDate.getDate() - 30); } const startMonth = Utilities.formatDate(startDate, 'Asia/Jakarta', 'yyyy-MM'); const endMonth = Utilities.formatDate(endDate, 'Asia/Jakarta', 'yyyy-MM'); const idx = getPanelIndex(); const panel = idx[panelId]; if (!panel || !panel.months) { return { status: 'success', data: [], stats: {}, panelId: panelId }; } const months = panel.months .filter(m => m.month >= startMonth && m.month <= endMonth) .map(m => m.month); const timeline = []; months.forEach(m => { const info = getPanelSheet(panelId, m); if (!info || !info.sheet) return; try { const sheet = info.sheet; const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; const tsIdx = headers.indexOf('timestamp'); const cbIdx = headers.indexOf('cb_status'); if (tsIdx === -1 || cbIdx === -1) return; const values = sheet.getRange(2, 1, sheet.getLastRow() - 1, headers.length).getValues(); values.forEach(row => { const ts = new Date(String(row[tsIdx]).replace(' ', 'T') + '+07:00'); if (isNaN(ts.getTime())) return; if (ts < startDate || ts > endDate) return; const status = parseInt(row[cbIdx]); if (isNaN(status)) return; timeline.push({ t: ts.getTime(), s: status }); }); } catch (err) { console.warn('Skip ' + m + ': ' + err.message); } }); timeline.sort((a, b) => a.t - b.t); let tripCount = 0; let tripDurationMs = 0; let lastTripStart = null; let prevStatus = null; timeline.forEach(point => { if (point.s === 0 && prevStatus !== 0) { tripCount++; lastTripStart = point.t; } else if (point.s === 1 && prevStatus === 0 && lastTripStart !== null) { tripDurationMs += (point.t - lastTripStart); lastTripStart = null; } prevStatus = point.s; }); if (lastTripStart !== null && timeline.length > 0) { tripDurationMs += (timeline[timeline.length - 1].t - lastTripStart); } return { status: 'success', data: timeline, panelId: panelId, stats: { tripCount: tripCount, tripDurationMs: tripDurationMs, tripDurationMin: Math.round(tripDurationMs / 60000), totalPoints: timeline.length }, meta: { period: period, start: startDate.toISOString(), end: endDate.toISOString() } }; } // ============ UTIL ============ function jsonResponse(obj, code) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } // ============ TEST ============ function testWrite() { const sample = { "Current A": 12.34, "Current B": 11.23, "Current C": 10.56, "Voltage A-B": 380.50, "Voltage B-C": 379.80, "Voltage C-A": 381.20, "Active Power Total": 7500.25, "Frequency": 50.01, "daily_usage_kwh": 45.67, "cb_status": 1 };
